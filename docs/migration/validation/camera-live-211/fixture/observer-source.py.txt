class Binary:
    def __init__(self,data):self.data=data;self.position=0
    def read(self,size):
        require(size>=0 and self.position+size<=len(self.data),'truncated binary record')
        value=self.data[self.position:self.position+size];self.position+=size;return value
    def integer(self):return struct.unpack('>i',self.read(4))[0]
    def long(self):return struct.unpack('>q',self.read(8))[0]
    def utf(self):
        size=struct.unpack('>H',self.read(2))[0]
        # DataOutputStream.writeUTF uses modified UTF-8/CESU-8, not ordinary UTF-8.
        value=self.read(size).replace(b'\xc0\x80',b'\0').decode('utf-8','surrogatepass')
        return value.encode('utf-16','surrogatepass').decode('utf-16')
    def bounded_count(self,limit):
        count=self.integer();require(0<=count<=limit,'binary count exceeds schema limit');return count
    def end(self):require(self.position==len(self.data),'binary record contains trailing bytes')


def checked_binary(data,magics,limit):
    require(40<=len(data)<=limit,'binary record byte limit')
    outer=Binary(data);magic=outer.integer();require(magic in magics,'unsupported binary schema')
    length=outer.integer();require(length==len(data)-40,'binary payload length differs')
    payload=outer.read(length);checksum=outer.read(32);outer.end()
    require(hashlib.sha256(payload).digest()==checksum,'binary envelope checksum mismatch')
    return magic,Binary(payload)


def project_record(data,project_id):
    _,reader=checked_binary(data,{0x504c5031},16*1024*1024)
    identity,name,created=reader.utf(),reader.utf(),reader.long()
    require(identity==project_id and created>=0,'project identity/timestamp differs')
    rows=[]
    for _ in range(reader.bounded_count(20000)):
        row={'identity':reader.utf(),'sourceFileName':reader.utf(),'name':reader.utf(),'bytes':reader.long(),'sha256':reader.utf()}
        require(HEX.fullmatch(row['identity']) and HEX.fullmatch(row['sha256']),'invalid project source identity')
        rows.append(row)
    reader.end();require(len({row['identity'] for row in rows})==len(rows),'duplicate project source identities')
    require(len({row['sourceFileName'] for row in rows})==len(rows),'duplicate project source files')
    return {'id':identity,'createdAt':created,'photos':rows,'checksumValid':True}


def task_record(data,project_id):
    magic,reader=checked_binary(data,{0x50505031,0x50505032},32*1024*1024)
    configs=[];previous=''
    for _ in range(reader.bounded_count(1024)):
        row={'projectId':reader.utf(),'presetId':reader.utf(),'presetName':reader.utf(),'enabledSince':reader.long()}
        require(row['projectId']>previous,'duplicate/unsorted project configs');previous=row['projectId']
        if row['projectId']==project_id:configs.append(row)
    tasks=[];previous=''
    for _ in range(reader.bounded_count(20000)):
        row={key:reader.utf() for key in ['projectId','identity','presetId','presetName','sourceFileName','name']}
        row['sourceBytes']=reader.long()
        row.update({key:reader.utf() for key in ['status','resultFileName','error']});row['updatedAt']=reader.long()
        row['saveTarget']=reader.utf() if magic==0x50505032 else 'PHONE_ALBUM'
        row['key']=row['projectId']+':'+row['identity']+':'+row['presetId']
        require(row['key']>previous,'duplicate/unsorted task keys');previous=row['key']
        if row['projectId']==project_id:tasks.append(row)
    reader.end();return {'configs':configs,'tasks':tasks,'checksumValid':True}


def decode_live(data,project_id,camera_id):
    require(32<=len(data)<=32*1024*1024,'live envelope byte limit')
    outer=json.loads(data);payload=outer.get('payload');require(isinstance(payload,str),'live payload is not a string')
    require(sha(payload.encode('utf-8'))==outer.get('sha256'),'live envelope checksum differs')
    state=json.loads(payload)
    require(state.get('schema')==1 and state.get('projectId')==project_id and state.get('cameraId')==camera_id,'live identity differs')
    entries={};pages=[state.get('entries',{})]+state.get('entryPages',[])
    require(len(pages)<=21,'too many live entry pages')
    for page in pages:
        require(isinstance(page,dict) and len(page)<=1000,'invalid live entry page')
        for key,value in page.items():
            require(HEX.fullmatch(key) and key not in entries and value.get('objectKey')==key,'duplicate/invalid live object identity')
            entries[key]=value
    require(len(entries)<=20000,'too many live entries')
    state['entries']=entries;state.pop('entryPages',None)
    return state


def jpeg_record(data,expected_sha,expected_bytes):
    require(len(data)==expected_bytes and sha(data)==expected_sha,'JPEG original/output bytes or SHA differs')
    with Image.open(io.BytesIO(data)) as image:
        require(image.format=='JPEG','file does not decode as JPEG');image.load();size=list(image.size)
    return {'bytes':len(data),'sha256':expected_sha,'decodedDimensions':size}
